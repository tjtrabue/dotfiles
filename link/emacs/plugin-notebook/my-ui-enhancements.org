Emacs, while infinitely customizable, offers little in the way of its default
UI. These plugins greatly enhance the user's editing experience by adding
beautiful visual components.

* sideline
=sideline= provides a universal diagnostics UI for the left and right sides of the
Emacs frame. Many other plugins present their own variants of this interface,
but none come /a la carte/ like =sideline=.

#+begin_src emacs-lisp
  (use-package sideline
    :delight (sideline-mode nil sideline)
    :hook
    ((flycheck-mode lsp-bridge-mode) . sideline-mode)
    :custom
    ;; Whether to display a left-diagnostic on the current line.
    (sideline-backends-left-skip-current-line nil)
    ;; Whether to display a right-diagnostic on the current line.
    (sideline-backends-right-skip-current-line nil)
    ;; Either 'up or 'down
    (sideline-order-left 'down)
    ;; Either 'up or 'down
    (sideline-order-right 'up)
    ;; Format string for left-diagnostics.
    (sideline-format-left "%s   ")
    ;; Format string for right-diagnostics.
    (sideline-format-right "   %s")
    ;; The overlay's priority.
    (sideline-priority 100)
    ;; Whether to display the backend's name along with the diagnostic.
    (sideline-display-backend-name t)
    ;; An alternative to `sideline-truncate'.
    ;; By default, sideline will not display a message if it is too long to display on the screen.
    ;; This option forces sideline to display the message even if it will not fit.
    ;; (sideline-force-display-if-exceeds t)
    ;; An alternative to `sideline-force-display-if-exceeds'.
    ;; Force `sideline' to display all messages but truncate them if they are too long to fit on the screen.
    (sideline-truncate t)
    ;; Sideline Backends (added one-by-one in each backend's configuration)
    (sideline-backends-right nil)
    (sideline-backends-left nil))
#+end_src

** Plugins
*** sideline-flycheck
Integrates =sideline= with Flycheck.

#+begin_src emacs-lisp
  (use-package sideline-flycheck
    :after sideline
    :hook (flycheck-mode . sideline-flycheck-setup)
    :init
    (add-to-list 'sideline-backends-right 'sideline-flycheck)
    (add-hook 'sideline-mode-hook
              (lambda ()
                (if (bound-and-true-p sideline-mode)
                    ;; Do not show Flycheck errors/diagnostics in the echo area since we want to hand
                    ;; over the display responsibilities to `sideline'.
                    (setq flycheck-display-errors-function nil)
                  ;; If we are leaving sideline-mode, return flycheck's error displaying function to
                  ;; its original value.
                  (setq flycheck-display-errors-function #'flycheck-display-error-messages)))))
#+end_src

*** sideline-eros
Display EROS results with =sideline=.

#+begin_src emacs-lisp
  (use-package sideline-eros
    :after sideline
    :straight
    (:type git :host github :repo "emacs-sideline/sideline-eros" :branch "master")
    :hook (sideline-mode . sideline-eros-setup)
    :init
    (add-to-list 'sideline-backends-right 'sideline-eros))
#+end_src

*** sideline-lsp
Display =lsp-mode= diagnostics in =sideline=.

#+begin_src emacs-lisp
  (use-package sideline-lsp
    ;; Not currently using `lsp-mode', and this plugin does not work with
    ;; `lsp-bridge-mode'.
    :disabled
    :after sideline
    :init
    (add-to-list 'sideline-backends-left 'sideline-lsp))
#+end_src
