Emacs has extensive support for editing, displaying, and exporting spreadsheets.

* csv-mode
Major mode for editing CSV (Comma Separated Values) files. Although files with
rows of comma-separated values with the =.csv= extension are most common, this
major mode works just as well with tab-separated value files (=.tsv= files) and
pipe-separated value files (=.psv= files). I highly recommend you activate both
=csv-mode= and the =csv-align-mode= minor mode automatically when editing CSV
files. =csv-align-mode= automatically keeps the CSV buffer's columns aligned.

#+begin_src emacs-lisp
  (use-package csv-mode
    :preface
    (require 'color)

    (defun my/csv-highlight (&optional separator)
      "Set a different color for each CSV column.

  An optional SEPARATOR char may be provided, if the file's separator is
  not a comma. The separator character may also be provided interactively,
  if called with a prefix argument.

  Taken from: https://www.emacswiki.org/emacs/CsvMode"
      (interactive (list (when current-prefix-arg (read-char "Separator: "))))
      (font-lock-mode 1)
      (let* ((separator (or separator ?\,))
             (n (count-matches (string separator) (pos-bol) (pos-eol)))
             (colors (cl-loop for i from 0 to 1.0 by (/ 2.0 n)
                              collect (apply #'color-rgb-to-hex
                                             (color-hsl-to-rgb i 0.3 0.5)))))
        (cl-loop for i from 2 to n by 2
                 for c in colors
                 for r = (format "^\\([^%c\n]+%c\\)\\{%d\\}" separator separator i)
                 do (font-lock-add-keywords nil `((,r (1 '(face (:foreground ,c)))))))))

    (defun my/csv-move-column-right (offset)
      "Move column at point OFFSET (defaults to 1) indices to the right."
      (interactive "p")
      (let* ((offset (or offset 1))
             (num-fields (1+ (count-matches "," (point-at-bol) (point-at-eol))))
             (index (csv--field-index))
             (new-index (+ index offset)))
        (when (> new-index num-fields)
          (setq new-index offset))
        (csv-kill-fields (list index) (point-min) (point-max))
        (csv-yank-fields new-index (point-min) (point-max))
        (if (<= (+ index offset) num-fields)
            (csv-forward-field (1+ offset))
          (csv-backward-field num-fields))
        (csv-beginning-of-field)))

    (defun my/csv-move-column-left (offset)
      "Move column at point OFFSET (defaults to 1) indices to the left."
      (interactive "p")
      (let* ((offset (or offset 1))
             (num-fields (1+ (count-matches "," (point-at-bol) (point-at-eol))))
             (index (csv--field-index))
             (new-index (- index offset)))
        (when (<= new-index 0)
          (setq new-index (1- new-index)))
        (csv-kill-fields (list index) (point-min) (point-max))
        (csv-yank-fields new-index (point-min) (point-max))
        (if (> (- index offset) 0)
            (csv-backward-field (1+ offset))
          (csv-forward-field num-fields)
          (csv-beginning-of-field))))
    :custom
    ;; All valid separator chars.
    ;; I include the pipe character "|" since .psv files are pipe-separated.
    (csv-separators '("," "	" "|"))
    :general
    (general-def csv-mode-map
      "M-h" 'my/csv-move-column-left
      "M-l" 'my/csv-move-column-right)
    :init
    ;; Open *.csv, *.psv, and *.tsv file buffers in `csv-mode'.
    ;; CSV - Comma-Separated Values
    ;; PSV - Pipe-Separated Values
    ;; TSV - Tab-Separated Values
    (dolist (e '(("\\.[Pp][Ss][Vv]\\'" . csv-mode)
                 ("\\.[Tt][Ss][Vv]\\'" . csv-mode)))
      (add-to-list 'auto-mode-alist e))
    (add-hook 'csv-mode-hook (lambda ()
                               ;; Align columns on the fly.
                               (csv-align-mode 1)
                               ;; Do not fold long lines in CSV files, as that would be confusing to
                               ;; read.
                               (toggle-truncate-lines nil)
                               ;; Set different colors for each column.
                               (my/csv-highlight))))
#+end_src
